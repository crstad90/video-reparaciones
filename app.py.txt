# --- RUTA 1: La página web con la Galería de Videos ---
@app.route('/')
def inicio():
    if os.path.exists(CARPETA_VIDEOS):
        videos = sorted([f for f in os.listdir(CARPETA_VIDEOS) if f.endswith('.mp4')], reverse=True)
    else:
        videos = []
    return render_template_string(HTML_PAGINA, videos=videos, total_videos=len(videos))

# --- RUTA 2: El disparador de la IA (Llamado cada 30 min por Cron-Job) ---
@app.route('/generar')
def disparar_generacion():
    try:
        termino_busqueda = random.choice(APARATOS_ELECTRONICOS)
        print(f"🤖 IA buscando imágenes para: {termino_busqueda}")
        
        urls_imagenes = [
            "https://unsplash.com", 
            "https://unsplash.com", 
            "https://unsplash.com"
        ]
        
        clips_de_imagen = []
        for i, url in enumerate(urls_imagenes):
            img_data = requests.get(url).content
            ruta_temporal = f"temp_img_{i}.jpg"
            with open(ruta_temporal, 'wb') as handler:
                handler.write(img_data)
                
            clip = ImageClip(ruta_temporal).set_duration(3)
            clips_de_imagen.append(clip)
            
        video_final = concatenate_videoclips(clips_de_imagen, method="compose")
        
        nombre_video = f"video_{int(time.time())}.mp4"
        ruta_guardado = os.path.join(CARPETA_VIDEOS, nombre_video)
        
        video_final.write_videofile(ruta_guardado, fps=24, codec="libx264")
        video_final.close()
        
        for i in range(len(urls_imagenes)):
            if os.path.exists(f"temp_img_{i}.jpg"):
                os.remove(f"temp_img_{i}.jpg")
                
        return jsonify({"status": "success", "video_creado": nombre_video}), 200
